\begin{abstract}
\noindent Ransomwares com criptografia intermitente (\emph{First-N} / \emph{skip-step}) destroem cabe\c{c}alhos, logs circulares e a \texttt{BAT} de cont\^{e}ineres \texttt{[MS-VHDX]} em minutos, deixando grande fra\c{c}\~{a}o dos blocos de \texttt{32 MiB} em texto claro. Recuperadores convencionais esgotam a RAM de resgate, incorrem em custo quadr\'{a}tico $O(K \cdot M \cdot n)$ ao revalidar \texttt{CRC-32C} com semente por inode em \texttt{ext4}, alucinam sobre colis\~{o}es JSON, \emph{extents unwritten} ou empates na \texttt{BAT}, e corrompem colmeias \texttt{regf} sujas ou o Active Directory (\texttt{NTDS.dit}). Formalizamos a m\'{a}quina determin\'{\i}stica de passagem \'{u}nica e mem\'{o}ria limitada $\mathcal{E}_{\mathrm{Rec}} = (\mathcal{V}_{\mathrm{src}}, B_{\mathrm{ram}}, \mathcal{F}, \mathcal{O}, \mathcal{G}_{\mathrm{veto}}, \mathcal{P}_{\mathrm{prov}}, \mathcal{V}_{\mathrm{out}})$ e avaliamos sua implementa\c{c}\~{a}o congelada (\texttt{v3.0.6} C\#/PS e Python 3 stdlib) nas baterias \texttt{B1}--\texttt{B6}: (1)~desacoplamento linear \texttt{GF(2)} de \texttt{CRC-32C} (\texttt{ext4} + \texttt{iSCSI} afim) com \texttt{0/10.000} diverg\^{e}ncias e speedup \texttt{168,73x} (\texttt{IC 95\% [144,58x, 193,88x]}); (2)~passagem \'{u}nica sem \emph{seek-back} na \texttt{BAT} (\texttt{260,32--269,34 MiB} WS CLR reconciliado com \texttt{221,18 MiB} anal\'{\i}tico a \texttt{100 GiB}); (3)~contabilidade 4-vias \texttt{ext4} (\texttt{0/200} erros, Multi-FS \texttt{F1=1,000}, \texttt{0/25} FP JSON); (4)~\texttt{SSAP/MCMF} bipartido com margem residual SPFA em componente $n \le 8192$ (\texttt{0/60} FAR, Wilson \texttt{95\% [0, 0,0602]}); e (5)~bloqueio de \texttt{20/20} \texttt{regf} sujas e \texttt{30/30} vetos Pr\'{e}-FS. Reportamos com transpar\^{e}ncia duas lacunas abertas na \texttt{v3.0.6}: salto silencioso em erro de leitura de \texttt{eh\_depth > 0} (\texttt{L10747} $\to$ \texttt{M-FUT-1}) e concord\^{a}ncia parcial \texttt{25/30} ($\kappa=0,6667$ $\to$ \texttt{M-FUT-2}) em \texttt{HardVetoes}, sob 28 refer\^{e}ncias prim\'{a}rias verificadas.
\end{abstract}

\section{Introdu\c{c}\~{a}o e Demarca\c{c}\~{a}o com \texttt{VTI-RES-2026-04}}

A criptografia intermitente~\cite{R07,R08,R09} destr\'{o}i metadados \texttt{[MS-VHDX]}~\cite{R16} deixando blocos de payload desconectados. Enquanto \texttt{VTI-RES-2026-04 v3.0} apresenta apenas a interface sist\^{e}mica \texttt{O1}--\texttt{O4}, este estudo aut\^{o}nomo formaliza $\mathcal{E}_{\mathrm{Rec}}$, demonstra \texttt{T1}--\texttt{T2} (escopado) e \texttt{P3}--\texttt{P4}, executa \texttt{B1}--\texttt{B6} em \texttt{os.tmpdir()} e documenta abertamente \texttt{M-FUT-1} e \texttt{M-FUT-2}.

\section{Tupla Formal $\mathcal{E}_{\mathrm{Rec}}$ e Seis Invariantes ($I_{R1}$--$I_{R6}$)}

\begin{equation}
\mathcal{E}_{\mathrm{Rec}} = \bigl(\mathcal{V}_{\mathrm{src}}, B_{\mathrm{ram}}, \mathcal{F}, \mathcal{O}, \mathcal{G}_{\mathrm{veto}}, \mathcal{P}_{\mathrm{prov}}, \mathcal{V}_{\mathrm{out}}\bigr)
\end{equation}
com sete operadores $\mathcal{O} = (O_{\mathrm{Log}}, O_{\mathrm{Pass1}}, O_{\mathrm{GF2}}, O_{\mathrm{Probe}}, O_{\mathrm{MCMF}}, O_{\mathrm{Surg}}, O_{\mathrm{Pass2}})$ regidos por $I_{R1}$ (evid\^{e}ncia somente-leitura~\cite{R15}), $I_{R2}$ (passagem \'{u}nica sem \emph{seek-back} na \texttt{BAT}), $I_{R3}$ (conserva\c{c}\~{a}o cont\'{a}bil 4-vias com \texttt{UnwrittenOnly}), $I_{R4}$ (equival\^{e}ncia bit-a-bit \texttt{GF(2)}), $I_{R5}$ (n\~{a}o-alucina\c{c}\~{a}o epist\^{e}mica limitando \texttt{BootReadinessScore} $\le 39$) e $I_{R6}$ (conserva\c{c}\~{a}o transacional \texttt{regf}).

\section{Resultados Te\'{o}ricos Principais (\texttt{T1}, \texttt{T2} Escopado, \texttt{P3}, \texttt{P4})}

\textbf{Teorema 1 (\texttt{T1} --- Linearidade \texttt{GF(2)} do \texttt{CRC-32C}):} Para o polin\^{o}mio Castagnoli~\cite{R01,R02,R03} com matriz refletida $A \in \mathbb{F}_2^{32\times 32}$,
\begin{equation}
\mathrm{CRC}_{\mathrm{ext4}}(s,X) = \mathrm{RawCrc32c}(0,X) \oplus A^{8n} s
\end{equation}
e no regime afim \texttt{iSCSI}/\texttt{[MS-VHDX]},
\begin{equation}
\mathrm{CRC}_{\mathrm{iSCSI}}(s_{\mathrm{init}},X) = \bigl(\mathrm{RawCrc32c}(0,X) \oplus \texttt{0xFFFFFFFF}\bigr) \oplus A^{8n} s_{\mathrm{init}}
\end{equation}
avaliado em $O(1)$ via 4 sub-tabelas de 256 palavras (\texttt{4 KiB})~\cite{R18}.

\textbf{Teorema 2 (\texttt{T2} --- Conserva\c{c}\~{a}o 4-Vias Escopada):} Sob legibilidade integral dos n\'{o}s de \'{\i}ndice \texttt{0xF30A}, \texttt{ee\_len > 32768} particiona \texttt{RawAlloc} em $\mathrm{EffAlloc} \sqcup \mathrm{UnwrittenOnly}$ sem contar \emph{unwritten} como ausente. \emph{Defeito aberto (\texttt{L10747} $\to$ \texttt{M-FUT-1}):} a \texttt{v3.0.6} ignora sub\'{a}rvores \texttt{eh\_depth > 0} ileg\'{\i}veis sem propagar \texttt{INDETERMINATE}.

\textbf{Proposi\c{c}\~{a}o 3 (\texttt{P3} --- Margem Residual \texttt{SSAP/MCMF}):} Em componente \'{u}nica $n \le 8192$~\cite{R04,R05,R06}, desfazer $S \to u \to p^* \to T$ e executar SPFA residual calcula exatamente a margem de ciclos alternantes $\Delta_u = \mathrm{OPT}(E \setminus \{(u,p^*)\}) - \mathrm{OPT}(E) \ge 0$, disparando \texttt{AMBIGUOUS\_BAT\_MAPPING} quando $\Delta_u < \Delta_{\min}$. Acima de \texttt{8192} n\'{o}s, o janelamento opera como \texttt{HEURISTIC\_WINDOWED}.

\textbf{Proposi\c{c}\~{a}o 4 (\texttt{P4} --- Cirurgia Transacional \texttt{regf}):} Mutar colmeias \texttt{regf}~\cite{R25} somente com \texttt{seq1 == seq2} preservados, \texttt{cbData} invariante, backup \texttt{DPAPI} e rec\'{a}lculo at\^{o}mico do checksum \texttt{XOR-32} (\texttt{508 B}) garante zero corrup\c{c}\~{a}o de colmeias sujas e zero deslocamento em \texttt{hbin}.

\section{Conclus\~{a}o e Declara\c{c}\~{a}o CRediT Adaptada (COPE / ACM / IEEE)}

$\mathcal{E}_{\mathrm{Rec}}$ une acelera\c{c}\~{a}o \texttt{GF(2)}, passagem \'{u}nica em RAM limitada, contabilidade 4-vias, veto \texttt{MCMF} residual e portas \texttt{regf}/Pr\'{e}-FS documentando abertamente \texttt{M-FUT-1} e \texttt{M-FUT-2}. Conforme COPE, ACM e IEEE, \textbf{Caio Gon\c{c}alves} figura como \textbf{\'{u}nico autor formal}, apoiado por rodadas n\~{a}o-autorais de revis\~{a}o adversarial e scripts determin\'{\i}sticos de benchmark.
