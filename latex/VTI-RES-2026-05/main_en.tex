\begin{abstract}
\noindent Intermittent encryption (\emph{First-N} / \emph{skip-step}) ransomware destroys \texttt{[MS-VHDX]} headers, circular logs, and Block Allocation Tables (\texttt{BAT}) in minutes while leaving most internal \texttt{32 MiB} payload blocks in plaintext. Conventional forensic utilities exhaust rescue RAM, suffer quadratic $O(K \cdot M \cdot n)$ cost re-hashing per-inode seeded \texttt{ext4} \texttt{CRC-32C} checksums, hallucinate over JSON magic collisions, \texttt{unwritten} extents, or symmetric \texttt{BAT} ties, and corrupt dirty \texttt{regf} hives or Active Directory (\texttt{NTDS.dit}) state. We formalize the deterministic single-pass, bounded-memory forensic machine $\mathcal{E}_{\mathrm{Rec}} = (\mathcal{V}_{\mathrm{src}}, B_{\mathrm{ram}}, \mathcal{F}, \mathcal{O}, \mathcal{G}_{\mathrm{veto}}, \mathcal{P}_{\mathrm{prov}}, \mathcal{V}_{\mathrm{out}})$ and evaluate its frozen implementation (\texttt{v3.0.6} C\#/PS and Python 3 stdlib) across six reproducible batteries (\texttt{B1}--\texttt{B6}): (1)~\texttt{GF(2)} linear decoupling of Castagnoli \texttt{CRC-32C} (\texttt{ext4} + affine \texttt{iSCSI}) achieving \texttt{0/10,000} divergences and \texttt{168.73x} mean speedup (\texttt{95\% CI [144.58x, 193.88x]}); (2)~single-pass read/write with zero \texttt{BAT} seek-back (\texttt{260.32--269.34 MiB} CLR WS reconciled with \texttt{221.18 MiB} analytical at \texttt{100 GiB}); (3)~4-way \texttt{ext4} conservation (\texttt{0/200} errors, Multi-FS \texttt{F1=1.000}, \texttt{0/25} JSON FPs); (4)~bipartite \texttt{SSAP/MCMF} with SPFA residual margin in single component $n \le 8192$ (\texttt{0/60} FAR, Wilson \texttt{95\% CI [0, 0.0602]}); and (5)~\texttt{20/20} dirty \texttt{regf} blocks and \texttt{30/30} Pre-FS hard vetoes. We transparently report two open boundaries in \texttt{v3.0.6}: silent read-failure skipping on \texttt{eh\_depth > 0} (\texttt{L10747} $\to$ \texttt{M-FUT-1}) and partial cross-OS \texttt{HardVetoes} agreement \texttt{25/30} ($\kappa=0.6667$ $\to$ \texttt{M-FUT-2}), grounded in 28 primary references.
\end{abstract}

\section{Main Theoretical Results (\texttt{T1}, Scoped \texttt{T2}, \texttt{P3}, \texttt{P4})}

\textbf{Theorem 1 (\texttt{T1} --- \texttt{GF(2)} Linear \texttt{CRC-32C}):} For Castagnoli~\cite{R01,R02,R03} with reflected transition matrix $A \in \mathbb{F}_2^{32\times 32}$,
\begin{equation}
\mathrm{CRC}_{\mathrm{ext4}}(s,X) = \mathrm{RawCrc32c}(0,X) \oplus A^{8n} s
\end{equation}
and in the affine \texttt{iSCSI}/\texttt{[MS-VHDX]} regime,
\begin{equation}
\mathrm{CRC}_{\mathrm{iSCSI}}(s_{\mathrm{init}},X) = \bigl(\mathrm{RawCrc32c}(0,X) \oplus \texttt{0xFFFFFFFF}\bigr) \oplus A^{8n} s_{\mathrm{init}}
\end{equation}
evaluated in $O(1)$ via four 256-word tables (\texttt{4 KiB})~\cite{R18}.

\textbf{Theorem 2 (\texttt{T2} --- Scoped 4-Way Conservation):} When all traversed \texttt{0xF30A} extent-tree index nodes are readable, \texttt{ee\_len > 32768} cleanly partitions \texttt{RawAlloc} into $\mathrm{EffAlloc} \sqcup \mathrm{UnwrittenOnly}$ with zero unwritten extents counted as missing. \emph{Open defect (\texttt{L10747} $\to$ \texttt{M-FUT-1}):} \texttt{v3.0.6} skips unreadable \texttt{eh\_depth > 0} subtrees without emitting \texttt{INDETERMINATE}.

\textbf{Proposition 3 (\texttt{P3} --- Residual \texttt{SSAP/MCMF} Margin):} Inside single component $n \le 8192$~\cite{R04,R05,R06}, unrouting $S \to u \to p^* \to T$ and running residual SPFA computes exact global alternating-cycle margin $\Delta_u = \mathrm{OPT}(E \setminus \{(u,p^*)\}) - \mathrm{OPT}(E) \ge 0$, firing \texttt{AMBIGUOUS\_BAT\_MAPPING} whenever $\Delta_u < \Delta_{\min}$. Above \texttt{8192} nodes, \texttt{v3.0.6} operates explicitly as \texttt{HEURISTIC\_WINDOWED}.

\textbf{Proposition 4 (\texttt{P4} --- Transactional \texttt{regf} Surgery):} Mutating \texttt{regf}~\cite{R25} strictly under \texttt{seq1 == seq2}, invariant \texttt{cbData}, \texttt{DPAPI} sibling backup, preserved \texttt{seq1 == seq2}, and atomic \texttt{508 B} \texttt{XOR-32} recomputation guarantees zero dirty-hive corruption and zero \texttt{hbin} cell shift.
