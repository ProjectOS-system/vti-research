\begin{abstract}
\noindent Opera\c{c}\~{o}es modernas de RaaS combinam explora\c{c}\~{a}o em appliances de borda sem agentes, intrus\~{o}es \emph{malware-free}, BYOVD, abuso de RMM comercial, dupla extors\~{a}o, criptografia intermitente \texttt{[MS-VHDX]} e destrui\c{c}\~{a}o acoplada de backups. Em contrapartida, arquiteturas puramente \emph{fail-closed} ou que acoplam LLMs ao bloqueio introduzem \emph{Self-DoS} e \emph{Indirect Prompt Injection}. Sob \emph{Design Science Research} estratificada (\texttt{E1}--\texttt{E4}), formalizamos a 8-tupla $\mathcal{E} = (\mathcal{S}_{\mathrm{Edge}}, \mathcal{S}_{\mathrm{Virt}}, \mathcal{S}_{\mathrm{Mesh}}, \mathcal{S}_{\mathrm{Ctrl}}, \mathcal{S}_{\mathrm{Desk}}, \mathcal{S}_{\mathrm{Vault}}, \mathcal{S}_{\mathrm{Clean}}, \mathcal{S}_{\mathrm{Cog}}; \mathcal{L}, \mathcal{T})$, treze hip\'{o}teses ($H_1$--$H_{13}$), treze controles \emph{fail-bounded} ($C_1$--$C_{13}$) e oito invariantes ($I_1$--$I_8$). A edi\c{c}\~{a}o \texttt{v3.0} verifica em c\'{o}digo execut\'{a}vel (\texttt{Bastion OS v1.3.0}: \texttt{301/301} testes backend, \texttt{254/254} E2E, \texttt{B1}--\texttt{B20}): (i)~conten\c{c}\~{a}o remota Zero-Inbound \texttt{BASTION-FLEET-CMD-V3} (\texttt{Bastion Approver} \emph{hardware-bound} \texttt{2-de-3}, sequenciador \texttt{ES256} TPM/CNG, barreira \texttt{START\_EXECUTION}, $B_{\mathrm{dom}} \ge Q_{\min}+1$, fallback $\le 90\text{ s}$); (ii)~Copiloto de Triagem IA Cont\'{\i}nua estritamente fora da TCB ($I_6$) sob LGPD Zero-Egress; (iii)~cofres \emph{Pull-Only} \texttt{BVLT1}/\texttt{BVPK1} com Google Drive Dual-Auth (\texttt{HTTP 308}/\texttt{206} + \texttt{pull-gdrive}) e Higiene Anti-\texttt{ENOSPC} $O(1)$; e (iv)~interface \texttt{O1}--\texttt{O4} demarcada da monografia aut\^{o}noma \texttt{VTI-RES-2026-05}, sob 63 refer\^{e}ncias verificadas.
\end{abstract}

\section{Introdu\c{c}\~{a}o e Estratifica\c{c}\~{a}o Epist\^{e}mica}

Campanhas destrutivas exploram quatro assimetrias: sabotagem de virtualiza\c{c}\~{a}o fora de hor\'{a}rio~\cite{R01,R18,R22}, criptografia intermitente \texttt{[MS-VHDX]}~\cite{R13,R14,R34}, acoplamento de raio de explos\~{a}o entre produ\c{c}\~{a}o, diret\'{o}rio, controle e backups \emph{push}~\cite{R02,R05,R59,R60}, e \emph{Indirect Prompt Injection} / vazamento LGPD em copilotos SOC~\cite{R54,R55,R57,R58}. Aplicamos proporcionalidade epist\^{e}mica ($\mathrm{Escopo}(\mathrm{Alegacao}) \le \mathrm{Escopo}(\mathrm{Evidencia})$) separando verifica\c{c}\~{a}o execut\'{a}vel isolada (\texttt{E1}: \texttt{Bastion OS v1.3.0}, \texttt{301/301} backend, \texttt{254/254} E2E, \texttt{B1}--\texttt{B20}), observa\c{c}\~{a}o forense anonimizada ($\mathcal{I}_0, N=1$, \texttt{E2}), propriedades formais condicionais (\texttt{E3}) e homologa\c{c}\~{a}o longitudinal em produ\c{c}\~{a}o (\texttt{E4}: \texttt{60 min} \texttt{G-PERF} e \texttt{SHADOW} $N_{\mathrm{blk}} \ge 300$).

\section{Modelo $\mathcal{E}$ e Oito Invariantes ($I_1$--$I_8$)}

\begin{equation}
\mathcal{E} = \bigl(\mathcal{S}_{\mathrm{Edge}}, \mathcal{S}_{\mathrm{Virt}}, \mathcal{S}_{\mathrm{Mesh}}, \mathcal{S}_{\mathrm{Ctrl}}, \mathcal{S}_{\mathrm{Desk}}, \mathcal{S}_{\mathrm{Vault}}, \mathcal{S}_{\mathrm{Clean}}, \mathcal{S}_{\mathrm{Cog}};\; \mathcal{L}, \mathcal{T}\bigr)
\end{equation}
com seis zonas disjuntas garantindo $\mathcal{K}_{\mathrm{TCB}} \cap \mathcal{K}(\mathcal{S}_{\mathrm{Cog}}) = \emptyset$, particionadas em invariantes locais ($I_1^{\mathrm{loc}}$ fencing monot\^{o}nico, $I_2$ transi\c{c}\~{a}o conservadora a \texttt{RESTRICTED\_RECOVERY}, $I_5$ autonomia offline limitada, $I_6$ exclus\~{a}o criptogr\'{a}fica de $\mathcal{S}_{\mathrm{Cog}}$, $I_7$ rollback local) e distribu\'{\i}dos ($I_1^{\mathrm{dist}}$ or\c{c}amento $B_{\mathrm{dom}}=1$ + qu\'{o}rum \texttt{2-de-3}, $I_3$ sobreviv\^{e}ncia \texttt{STAGING\_RESTORE}, $I_4$ qu\'{o}rum versionado $\ge Q_{\min}+1$ anti-ABA~\cite{R60}, $I_8$ continuidade \emph{War-Room Offline} \texttt{SQLCipher 4}).

\section{Treze Controles \emph{Fail-Bounded} ($C_1$--$C_{13}$)}

\textbf{$C_1$:} Capability \texttt{Ed25519} \'{u}nica, \texttt{EvtSubscribe} + 14 regras \texttt{Sigma} e \texttt{DiskHygieneEngine} $O(1)$ (6 travas inviol\'{a}veis, \texttt{Tier 3} din\^{a}mico, rollback \texttt{SHA-256})~\cite{R46,R47,R53}.
\textbf{$C_2$:} Barramento Zero-Inbound \texttt{BASTION-FLEET-CMD-V3} (\texttt{F3}--\texttt{F7}) com 5 chaves \texttt{ES256} em TPM/CNG (proibindo \texttt{SOFTWARE\_KEY} em produ\c{c}\~{a}o), \texttt{Bastion Approver} Rust (\texttt{UP=1+UV=1}, \emph{hardware-bound} \texttt{2-de-3}), barreira \texttt{START\_EXECUTION} + journals \texttt{PRAGMA synchronous = FULL}, isolamento IPv4/IPv6/QUIC com fallback $\le 90\text{ s}$, $B_{\mathrm{dom}} \ge Q_{\min}+1$ e executor Hyper-V n\~{a}o-destrutivo proibindo \texttt{Remove-VM}~\cite{R04,R41,R44,R45,R49,R59,R61,R62}.
\textbf{$C_3$:} Malha Viva Anti-\texttt{TOCTOU}, atribui\c{c}\~{a}o SMB/NFS, guardi\~{a}o \texttt{[MS-VHDX]} $< 1\text{ MiB}$ e interface sist\^{e}mica \texttt{O1}--\texttt{O4} remetendo a teoria alg\'{e}brica fechada a \texttt{VTI-RES-2026-05}.
\textbf{$C_4$:} Cofre \emph{Pull-Only} (\texttt{nic\_prod} \texttt{100\% SYN DROP}), \texttt{BVLT1}/\texttt{BVPK1} (\texttt{scrypt}, \texttt{HKDF}, \texttt{AES-256-GCM}, \texttt{FASTCDC\_NC2}, \texttt{HTTP 206} estrito), \texttt{DR Zero-Touch} e \texttt{Google Drive Dual-Auth} (\texttt{JWT RS256} \texttt{canDelete=false} e \texttt{PKCE S256} \texttt{drive.file}, \texttt{HTTP 308}, \texttt{700 GiB/24h}, \texttt{pull-gdrive})~\cite{R02,R33,R35,R36,R37,R38,R50,R51}.
\textbf{$C_5$--$C_7$:} Journal \texttt{TPM 2.0} em 2 fases + \texttt{NTS}/\texttt{RFC 3161} + Testemunha Anti-Rollback; \emph{Clean-Room DR} dupla trilha (\texttt{Ed25519}); e fila $R_{\mathrm{VTI}} \in [0,100]$.
\textbf{$C_8$:} Copiloto IA Cont\'{\i}nua (\texttt{Onda 7-b}) fora da TCB ($I_6$) com roteamento 4-Tiers, compressor Ponytail, cercas anti-envenenamento, schema 7-chaves, 5 ferramentas \emph{read-only} isoladas, consenso dual fail-safe (\texttt{BENIGN\_LIKELY} n\~{a}o-autoritativo) e \texttt{LGPD Zero-Egress} por padr\~{a}o~\cite{R54,R55,R56,R57,R58}.
\textbf{$C_9$--$C_{13}$:} Disjuntor de exfiltra\c{c}\~{a}o, manifesto RMM/t\'{u}neis, \texttt{WDAC} 4 an\'{e}is, coleta de borda + *honey-tokens*, \texttt{LOG\_GAP\_ALERT} anti-\texttt{ENOSPC}, \texttt{ZERO\_INSTANCE\_DATA\_BEFORE\_AUTH} e esta\c{c}\~{a}o *War-Room Offline* $\mathcal{S}_{\mathrm{Desk}}$~\cite{R29,R30,R31,R32}.
